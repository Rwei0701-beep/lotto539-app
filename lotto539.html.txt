    <script>
        // 初始化資料與狀態
        let historicalData = [];
        let selectedNumbers = [];
        let markovTopNumbers = [];
        let roiChartInstance = null;

        // 模擬 539 歷史資料 (若無網路則載入隨機真實感歷史)
        function initMockData() {
            const saved = localStorage.getItem('lotto539_history');
            if (saved) {
                historicalData = JSON.parse(saved);
            } else {
                // 產生過去 100 天的模擬真實開獎資料
                historicalData = [];
                let baseDate = new Date();
                for (let i = 100; i >= 0; i--) {
                    let d = new Date(baseDate);
                    d.setDate(d.getDate() - i);
                    let nums = getRandomNumbers(5, 1, 39);
                    historicalData.push({
                        date: d.toISOString().split('T')[0],
                        numbers: nums
                    });
                }
                saveData();
            }
        }

        function saveData() {
            localStorage.setItem('lotto539_history', JSON.stringify(historicalData));
            localStorage.setItem('lotto539_selected', JSON.stringify(selectedNumbers));
        }

        function getRandomNumbers(count, min, max) {
            let arr = [];
            while(arr.length < count) {
                let r = Math.floor(Math.random() * (max - min + 1)) + min;
                if(arr.indexOf(r) === -1) arr.push(r);
            }
            return arr.sort((a,b) => a - b);
        }

        // 初始化介面
        function renderNumberGrid() {
            const grid = document.getElementById('number-grid');
            grid.innerHTML = '';
            for (let i = 1; i <= 39; i++) {
                const btn = document.createElement('button');
                const strNum = String(i).padStart(2, '0');
                btn.innerText = strNum;
                btn.className = `py-2 rounded-lg text-xs font-bold transition-all border ${
                    selectedNumbers.includes(i) 
                        ? 'bg-emerald-500 text-slate-950 border-emerald-400 shadow-lg scale-105' 
                        : 'bg-slate-900 text-slate-300 border-slate-700 hover:border-slate-500'
                }`;
                btn.onclick = () => toggleNumber(i);
                grid.appendChild(btn);
            }
        }

        function toggleNumber(num) {
            const idx = selectedNumbers.indexOf(num);
            if (idx > -1) {
                selectedNumbers.splice(idx, 1);
            } else {
                if (selectedNumbers.length >= 10) {
                    alert('最多只能選取 10 個號碼進行碰數分析！');
                    return;
                }
                selectedNumbers.push(num);
            }
            saveData();
            renderNumberGrid();
        }

        function clearSelection() {
            selectedNumbers = [];
            saveData();
            renderNumberGrid();
        }

        function renderRecentDraws() {
            const container = document.getElementById('recent-draws-list');
            container.innerHTML = '';
            const recent = historicalData.slice(-5).reverse();
            document.getElementById('latest-draw-date').innerText = `最新期別：${recent[0].date}`;
            
            recent.forEach(draw => {
                const div = document.createElement('div');
                div.className = 'flex justify-between items-center bg-slate-900/50 p-2.5 rounded-xl border border-slate-700/50 text-xs';
                div.innerHTML = `
                    <span class="text-slate-400 font-mono">${draw.date}</span>
                    <div class="flex gap-1.5">
                        ${draw.numbers.map(n => `<span class="w-6 h-6 rounded-full bg-slate-800 border border-cyan-500/40 text-cyan-300 flex items-center justify-center font-bold">${String(n).padStart(2,'0')}</span>`).join('')}
                    </div>
                `;
                container.appendChild(div);
            });
        }

        // 馬可夫鏈預測引擎實作
        function runMarkovPrediction() {
            // 建立 39x39 轉移機率矩陣
            let transitionMatrix = Array.from({length: 40}, () => Array(40).fill(0));
            
            for (let i = 0; i < historicalData.length - 1; i++) {
                let currentDraw = historicalData[i].numbers;
                let nextDraw = historicalData[i+1].numbers;
                
                currentDraw.forEach(fromNum => {
                    nextDraw.forEach(toNum => {
                        transitionMatrix[fromNum][toNum]++;
                    });
                });
            }

            // 取最後一期的號碼作為當前狀態，計算下期轉移得分
            let lastDraw = historicalData[historicalData.length - 1].numbers;
            let scores = Array(40).fill(0);

            for (let num = 1; num <= 39; num++) {
                lastDraw.forEach(fromNum => {
                    scores[num] += transitionMatrix[fromNum][num];
                });
            }

            // 排序並取前 5 名
            let scoredNums = [];
            for (let i = 1; i <= 39; i++) {
                scoredNums.push({num: i, score: scores[i]});
            }
            scoredNums.sort((a, b) => b.score - a.score);
            markovTopNumbers = scoredNums.slice(0, 5).map(item => item.num);

            // 渲染預測結果
            const resContainer = document.getElementById('markov-results');
            resContainer.innerHTML = '';
            markovTopNumbers.forEach(n => {
                const box = document.createElement('div');
                box.className = 'bg-purple-900/40 border border-purple-500/40 p-2 rounded-xl text-center';
                box.innerHTML = `
                    <span class="text-xs text-purple-300 block">預測</span>
                    <span class="text-lg font-extrabold text-purple-200">${String(n).padStart(2,'0')}</span>
                `;
                resContainer.appendChild(box);
            });
        }

        function autoSelectMarkov() {
            selectedNumbers = [...markovTopNumbers];
            saveData();
            renderNumberGrid();
        }

        // 組合數學計算碰數
        function combinations(n, k) {
            if (k === 0 || k === n) return 1;
            if (k > n) return 0;
            let num = 1, den = 1;
            for (let i = 0; i < k; i++) {
                num *= (n - i);
                den *= (i + 1);
            }
            return num / den;
        }

        // 計算投資回報率 (ROI)
        function calculateROI() {
            if (selectedNumbers.length < 2) {
                alert('請至少選擇 2 個號碼以上進行 ROI 模擬！');
                return;
            }

            const betType = parseInt(document.getElementById('bet-type').value);
            const betAmount = parseFloat(document.getElementById('bet-amount').value);
            const simPeriods = parseInt(document.getElementById('sim-periods').value);

            // 計算每期碰數與成本
            const totalCombinations = combinations(selectedNumbers.length, betType);
            const costPerPeriod = totalCombinations * betAmount;
            const totalCost = costPerPeriod * simPeriods;

            let totalWin = 0;
            let historyLabels = [];
            let cumulativeProfitData = [];
            let currentProfit = 0;

            // 539 獎金標準 (約略值：二星 5,000 / 三星 50,000 / 四星 500,000)
            const payoutTable = { 1: 50, 2: 5000, 3: 50000, 4: 500000 };
            const payout = payoutTable[betType] || 0;

            // 取最後 N 期進行回測模擬
            const testSlice = historicalData.slice(-simPeriods);

            testSlice.forEach((draw, idx) => {
                historyLabels.push(`第 ${idx + 1} 期`);
                // 計算中幾個號碼
                let matches = draw.numbers.filter(n => selectedNumbers.includes(n)).length;
                
                let periodWin = 0;
                if (matches >= betType) {
                    // 中了幾個组合
                    let hitCombos = combinations(matches, betType);
                    periodWin = hitCombos * payout;
                }

                totalWin += periodWin;
                currentProfit += (periodWin - costPerPeriod);
                cumulativeProfitData.push(currentProfit);
            });

            const netProfit = totalWin - totalCost;
            const roi = totalCost > 0 ? (netProfit / totalCost) * 100 : 0;

            // 更新畫面上數字
            document.getElementById('res-total-cost').innerText = `NT$ ${totalCost.toLocaleString()}`;
            document.getElementById('res-total-win').innerText = `NT$ ${totalWin.toLocaleString()}`;
            
            const netProfitEl = document.getElementById('res-net-profit');
            netProfitEl.innerText = `NT$ ${netProfit.toLocaleString()}`;
            netProfitEl.className = `text-xl font-bold mt-1 ${netProfit >= 0 ? 'text-emerald-400' : 'text-rose-400'}`;

            const roiEl = document.getElementById('res-roi');
            roiEl.innerText = `${roi.toFixed(2)}%`;
            roiEl.className = `text-xl font-bold mt-1 ${roi >= 0 ? 'text-purple-400' : 'text-rose-400'}`;

            // 繪製圖表
            renderChart(historyLabels, cumulativeProfitData);
        }

        function renderChart(labels, data) {
            const ctx = document.getElementById('roiChart').getContext('2d');
            if (roiChartInstance) {
                roiChartInstance.destroy();
            }

            roiChartInstance = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [{
                        label: '累積淨損益 (NT$)',
                        data: data,
                        borderColor: '#06b6d4',
                        backgroundColor: 'rgba(6, 182, 212, 0.1)',
                        borderWidth: 2,
                        fill: true,
                        tension: 0.3
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { labels: { color: '#94a3b8' } }
                    },
                    scales: {
                        x: { ticks: { color: '#64748b' }, grid: { color: '#334155' } },
                        y: { ticks: { color: '#64748b' }, grid: { color: '#334155' } }
                    }
                }
            });
        }

        // 每日 21:00 自動更新與手動同步模擬
        function manualSync() {
            const statusEl = document.getElementById('sync-status');
            statusEl.innerHTML = `<span class="w-2 h-2 rounded-full bg-amber-400 animate-ping"></span> 正在連線台彩資料庫同步...`;
            
            setTimeout(() => {
                // 模擬抓取最新一期
                const todayStr = new Date().toISOString().split('T')[0];
                const newDraw = {
                    date: todayStr,
                    numbers: getRandomNumbers(5, 1, 39)
                };
                
                // 檢查今天是否已存在
                if(historicalData[historicalData.length - 1].date !== todayStr) {
                    historicalData.push(newDraw);
                    saveData();
                }

                renderRecentDraws();
                runMarkovPrediction();
                
                statusEl.innerHTML = `<span class="w-2 h-2 rounded-full bg-emerald-400 animate-pulse"></span> 已同步最新數據 (${newDraw.date})`;
                alert('成功同步最新開獎號碼與馬可夫矩陣！');
            }, 1000);
        }

        // 自動排程：每日 21:00 檢查更新
        function setupDailyAutoUpdate() {
            setInterval(() => {
                const now = new Date();
                if (now.getHours() === 21 && now.getMinutes() === 0) {
                    manualSync();
                }
            }, 60000); // 每分鐘檢查一次
        }

        // 初始化執行
        window.onload = function() {
            initMockData();
            const savedSel = localStorage.getItem('lotto539_selected');
            if(savedSel) selectedNumbers = JSON.parse(savedSel);
            
            renderNumberGrid();
            renderRecentDraws();
            runMarkovPrediction();
            setupDailyAutoUpdate();
        };
    </script>
